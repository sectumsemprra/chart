import os

baselines = {
    "GRPO": "./checkpoints/grpo-100-seed2025",  # Or wherever your GRPO checkpoint is
    "NSR": "./checkpoints/nsr-100-seed2025",    # Or wherever your NSR checkpoint is
    "PSR": "./checkpoints/psr-100-seed2025",
    "W-REINFORCE": "./checkpoints/wreinforce-100-seed2025",
}

all_good = True
for name, path in baselines.items():
    # Try to find checkpoint
    checkpoint_found = False

    # Check exact path
    if os.path.exists(path):
        checkpoint_found = True

    # Check if checkpoint exists anywhere
    if not checkpoint_found:
        for root, dirs, files in os.walk('.'):
            if name.lower().replace('-', '') in root.lower():
                if 'checkpoint' in root and any('adapter_model' in f for f in files):
                    path = root
                    checkpoint_found = True
                    break

    if checkpoint_found:
        print(f"✓ {name:15s} {path}")
    else:
        print(f"✗ {name:15s} NOT FOUND")
        all_good = False

print("="*80)
if all_good:
    print("✓ ALL 4 BASELINES READY FOR EVALUATION!")
else:
    print("⚠ Some checkpoints missing - check paths above")
print("="*80)

from google.colab import drive
import shutil

# Mount Drive
drive.mount('/content/drive')

# Create backup directory
backup_dir = '/content/drive/MyDrive/thesis/checkpoints_100_samples'
os.makedirs(backup_dir, exist_ok=True)

# Copy all checkpoints
for name, path in baselines.items():
    if os.path.exists(path):
        dest = os.path.join(backup_dir, name.lower().replace('-', '_'))
        print(f"Copying {name}...")
        try:
            shutil.copytree(path, dest, dirs_exist_ok=True)
            print(f"  ✓ Saved to {dest}")
        except Exception as e:
            print(f"  ✗ Error: {e}")

# Copy logs
log_dir = '/content/drive/MyDrive/thesis/logs_100_samples'
os.makedirs(log_dir, exist_ok=True)

for log_file in ['grpo.log', 'nsr.log', 'psr_100.log', 'wreinforce_100.log']:
    if os.path.exists(log_file):
        dest = os.path.join(log_dir, log_file)
        shutil.copy(log_file, dest)
        print(f"✓ {log_file} → Drive")

print("\n" + "="*80)
print("✓ ALL CHECKPOINTS BACKED UP TO GOOGLE DRIVE")
print(f"Location: {backup_dir}")
